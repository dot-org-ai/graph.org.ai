# Generative Functions

Generative functions leverage large language models and AI systems to produce novel outputs, generate code, create structured data, and augment human capabilities through intelligent automation.

## Overview

Generative functions represent a new class of computational operations that harness the power of trained models to generate contextually relevant, semantically meaningful outputs. Unlike traditional deterministic functions, generative functions operate in probabilistic spaces, balancing creativity with guidance through prompt engineering, structured outputs, and fine-tuning.

## Large Language Models (LLMs)

Modern generative functions are powered by large language models trained on vast corpora of text data.

### Prompt Techniques

```typescript
// Few-shot prompting: provide examples
function fewShotPrompt(
  task: string,
  examples: { input: string; output: string }[],
  query: string
): string {
  const exampleText = examples
    .map((ex, i) => `Example ${i + 1}:\nInput: ${ex.input}\nOutput: ${ex.output}`)
    .join('\n\n')

  return `Task: ${task}\n\n${exampleText}\n\nNow solve:\nInput: ${query}`
}

// Chain of thought: make reasoning explicit
function chainOfThoughtPrompt(question: string): string {
  return `Q: ${question}\n\nLet me think through this step by step:\n1. First, I need to understand...`
}

// Role-based prompting: assign a persona
function roleBasedPrompt(role: string, task: string, query: string): string {
  return `You are a ${role}.\n\nTask: ${task}\n\nQuestion: ${query}`
}

// Instruction-based prompting: detailed instructions
function instructionBasedPrompt(
  goal: string,
  instructions: string[],
  input: string
): string {
  const instructionText = instructions
    .map((inst, i) => `${i + 1}. ${inst}`)
    .join('\n')

  return `Goal: ${goal}\n\nInstructions:\n${instructionText}\n\nInput:\n${input}`
}

// Retrieval-augmented prompting: include context
function ragPrompt(
  question: string,
  relevantDocuments: string[]
): string {
  const context = relevantDocuments
    .map((doc, i) => `[Document ${i + 1}]\n${doc}`)
    .join('\n\n')

  return `Context:\n${context}\n\nQuestion: ${question}`
}

// Constrained generation: specify format
function constrainedPrompt(
  task: string,
  format: string,
  query: string
): string {
  return `${task}\n\nFormat your response as:\n${format}\n\n${query}`
}
```

## Prompt Chaining and Workflows

Composing multiple generative calls into intelligent workflows.

```typescript
// Prompt chain: sequential generative calls
interface PromptChainStep {
  name: string
  prompt: string
  inputVariables: string[]
  outputVariable: string
  options?: GenerationOptions
}

interface PromptChainState {
  [key: string]: string | object
}

async function executePromptChain(
  steps: PromptChainStep[],
  initialState: PromptChainState
): Promise<PromptChainState> {
  let state = initialState

  for (const step of steps) {
    const filledPrompt = fillTemplate(step.prompt, state)
    const result = await generateText(filledPrompt, step.options)
    state[step.outputVariable] = result
  }

  return state
}

function fillTemplate(template: string, variables: Record<string, any>): string {
  return template.replace(/\{(\w+)\}/g, (match, key) => {
    return String(variables[key] ?? match)
  })
}

// Example: Multi-step analysis chain
const analysisChain: PromptChainStep[] = [
  {
    name: 'Extract Main Points',
    prompt: 'Extract the 3 main points from: {document}',
    inputVariables: ['document'],
    outputVariable: 'mainPoints'
  },
  {
    name: 'Analyze Sentiment',
    prompt: 'Analyze the sentiment of these points: {mainPoints}',
    inputVariables: ['mainPoints'],
    outputVariable: 'sentiment'
  },
  {
    name: 'Generate Summary',
    prompt: 'Summarize this analysis:\nPoints: {mainPoints}\nSentiment: {sentiment}',
    inputVariables: ['mainPoints', 'sentiment'],
    outputVariable: 'summary'
  }
]

// Tree of thought: exploring multiple reasoning paths
interface ThoughtNode {
  thought: string
  children: ThoughtNode[]
  score?: number
}

async function generateTreeOfThought(
  question: string,
  depth: number = 3
): Promise<ThoughtNode> {
  const root: ThoughtNode = {
    thought: question,
    children: []
  }

  async function expandNode(node: ThoughtNode, currentDepth: number) {
    if (currentDepth >= depth) return

    const prompt = `Given this question/thought: "${node.thought}"\n\nGenerate 3 diverse approaches to explore:`
    const thoughts = await generateText(prompt)

    for (const thought of thoughts.split('\n').slice(0, 3)) {
      const child: ThoughtNode = { thought, children: [] }
      node.children.push(child)
      await expandNode(child, currentDepth + 1)
    }
  }

  await expandNode(root, 0)
  return root
}
```

## Embeddings and Semantic Operations

Working with vector representations of text for similarity and search.

```typescript
// Embedding interface
interface EmbeddingResult {
  text: string
  embedding: number[]
  metadata?: Record<string, any>
}

// Similarity computation
function cosineSimilarity(v1: number[], v2: number[]): number {
  const dotProduct = v1.reduce((sum, a, i) => sum + a * v2[i], 0)
  const magnitude1 = Math.sqrt(v1.reduce((sum, x) => sum + x * x, 0))
  const magnitude2 = Math.sqrt(v2.reduce((sum, x) => sum + x * x, 0))
  return dotProduct / (magnitude1 * magnitude2)
}

// Embedding-based search
async function semanticSearch(
  query: string,
  documents: EmbeddingResult[],
  topK: number = 5
): Promise<EmbeddingResult[]> {
  const queryEmbedding = await generateEmbedding(query)

  const scored = documents.map(doc => ({
    ...doc,
    score: cosineSimilarity(queryEmbedding, doc.embedding)
  }))

  return scored
    .sort((a, b) => b.score - a.score)
    .slice(0, topK)
}

// Vector database operations
interface VectorStore {
  add(embeddings: EmbeddingResult[]): Promise<void>
  search(queryEmbedding: number[], topK: number): Promise<EmbeddingResult[]>
  delete(ids: string[]): Promise<void>
  update(id: string, embedding: EmbeddingResult): Promise<void>
}

// Semantic caching
class EmbeddingCache {
  private cache = new Map<string, EmbeddingResult>()

  async get(text: string): Promise<number[]> {
    if (this.cache.has(text)) {
      return this.cache.get(text)!.embedding
    }

    const result = await generateEmbedding(text)
    this.cache.set(text, { text, embedding: result })
    return result
  }

  clear() {
    this.cache.clear()
  }
}
```

## Performance and Cost Optimization

```typescript
// Token counting and estimation
function estimateTokenCount(text: string): number {
  // Approximate: 1 token per 4 characters for English
  return Math.ceil(text.length / 4)
}

interface CostEstimate {
  inputTokens: number
  outputTokens: number
  estimatedCost: number
}

function estimateCost(
  inputText: string,
  maxOutputTokens: number,
  model: string
): CostEstimate {
  const inputTokens = estimateTokenCount(inputText)
  // Pricing varies by model
  const inputCost = inputTokens * 0.0015 / 1000  // Example pricing
  const outputCost = maxOutputTokens * 0.002 / 1000

  return {
    inputTokens,
    outputTokens: maxOutputTokens,
    estimatedCost: inputCost + outputCost
  }
}

// Batch processing for cost efficiency
async function batchGenerate(
  prompts: string[],
  batchSize: number = 10
): Promise<string[]> {
  const results: string[] = []

  for (let i = 0; i < prompts.length; i += batchSize) {
    const batch = prompts.slice(i, i + batchSize)
    const batchResults = await Promise.all(
      batch.map(prompt => generateText(prompt))
    )
    results.push(...batchResults)
  }

  return results
}

// Caching generated outputs
class GenerationCache {
  private cache = new Map<string, string>()

  async generateOrCache(
    prompt: string,
    options?: GenerationOptions
  ): Promise<string> {
    const key = JSON.stringify({ prompt, options })

    if (this.cache.has(key)) {
      return this.cache.get(key)!
    }

    const result = await generateText(prompt, options)
    this.cache.set(key, result)
    return result
  }
}
```

## Best Practices

1. **Clear Prompts**: Write specific, detailed prompts with examples
2. **Constrained Output**: Use structured generation for reliable outputs
3. **Validation**: Always validate generated content before use
4. **Cost Management**: Monitor token usage and implement caching
5. **Error Handling**: Gracefully handle API failures and retries
6. **Security**: Never include sensitive data in prompts
7. **Version Control**: Track prompt changes and model versions
8. **Testing**: Test generation outputs across diverse inputs
9. **Monitoring**: Log and analyze generation quality and cost
10. **User Feedback**: Collect feedback to improve generation quality
