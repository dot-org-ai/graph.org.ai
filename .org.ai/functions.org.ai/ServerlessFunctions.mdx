# Serverless Functions

Serverless computing enables scaling applications without managing infrastructure. Serverless functions represent the paradigm shift from server management to event-driven, function-oriented architecture where organizations pay only for actual computation time.

## Overview

Serverless functions are event-driven code units that execute in managed environments with automatic scaling, built-in high availability, and per-invocation billing. Unlike traditional servers, developers don't manage capacity planning, operating system patching, or infrastructure provisioning. The abstraction layer handles resource allocation, networking, and fault tolerance transparently.

```typescript
// Runtime selection
function selectOptimalRuntime(
  requirements: FunctionRequirements,
  platforms: ServerlessPlatform[]
): { platform: ServerlessPlatform; runtime: Runtime } | null {
  const candidates = platforms.filter(p => {
    return (
      p.maxMemory >= requirements.memoryMB &&
      p.maxTimeout >= requirements.timeoutSeconds &&
      p.maxPayloadSize >= requirements.payloadSizeMB * 1024 * 1024
    )
  })

  // Score by cold start and cost
  const scored = candidates.map(platform => ({
    platform,
    runtime: platform.supportedRuntimes.find(r => r.language === requirements.language)!,
    coldStartScore: platform.coldStartMs,
    costScore: platform.pricingPerGBSecond * requirements.memoryMB / 1024
  }))

  const best = scored.reduce((best, current) =>
    (current.coldStartScore + current.costScore) < (best.coldStartScore + best.costScore)
      ? current
      : best
  )

  return { platform: best.platform, runtime: best.runtime! }
}

  timeoutSeconds: number
  payloadSizeMB: number
  expectedInvocations: number
```

```typescript
// Optimization strategies
class ColdStartOptimizer {
  optimizeImports(code: string): string {
    // Move require/import statements inside handlers
    // Use lazy loading for heavy dependencies
    return code
      .replace(
        /^const (\w+) = require\(['"]([^'"]+)['"]\);?$/gm,
        (match, varName, module) => {
          if (this.isHeavyModule(module)) {
            return `let ${varName};`
          }
          return match
        }
      )
  }

  private isHeavyModule(module: string): boolean {
    const heavyModules = ['aws-sdk', 'sharp', 'pandas', 'tensorflow']
    return heavyModules.some(heavy => module.includes(heavy))
  }

  createLazyLoader(moduleName: string): string {
    return `
      let ${moduleName};
      function load${moduleName}() {
        if (!${moduleName}) {
          ${moduleName} = require('${moduleName}');
        }
        return ${moduleName};
      }
    `
  }

  // Bundle and minify
  optimizeBundle(): string {
    return `
      // Use tree-shaking and code splitting
      // Minify production code
      // Remove dev dependencies
      // Use compression
    `
  }

  // Container image optimization
  createLightweightContainer(): string {

      FROM node:20-alpine
      WORKDIR /app
      COPY package*.json ./
```

```typescript
// Event handler patterns
async function httpHandler(event: APIGatewayProxyEvent): Promise<APIGatewayProxyResult> {
  try {
    const { path, httpMethod, body } = event
    const parsedBody = body ? JSON.parse(body) : null

    // Route based on method and path
    let result: any

    if (httpMethod === 'GET' && path === '/items') {
      result = await getItems()
    } else if (httpMethod === 'POST' && path === '/items') {
      result = await createItem(parsedBody)
    } else if (httpMethod === 'PUT' && path.match(/^\/items\/\d+$/)) {
      const id = path.split('/')[2]
      result = await updateItem(id, parsedBody)
    } else if (httpMethod === 'DELETE' && path.match(/^\/items\/\d+$/)) {
      const id = path.split('/')[2]
      result = await deleteItem(id)
    } else {
      return {
        statusCode: 404,
        body: JSON.stringify({ error: 'Not found' })
      }
    }

    return {
      statusCode: 200,
      body: JSON.stringify(result),
      headers: { 'Content-Type': 'application/json' }
    }
  } catch (error) {
    return {
      statusCode: 500,
      body: JSON.stringify({ error: String(error) })
    }
  }
}

async function dynamoDBStreamHandler(event: DynamoDBStreamEvent): Promise<void> {

// Async job processing with SQS
async function sqsHandler(event: SQSEvent): Promise<void> {
  const batchItemFailures = []

  for (const record of event.Records) {
    try {
      const message = JSON.parse(record.body)
      await processJob(message)
    } catch (error) {
      // Report failure for retry
      batchItemFailures.push({
        itemId: record.messageId
      })
    }
  }

  // Return failed messages to queue
  return { batchItemFailures }
}
```

```typescript
// Cost calculator
class CostCalculator {
  calculateMonthlyCost(
    invocations: number,
    avgDurationMs: number,
    memoryMB: number,
    platform: ServerlessPlatform
  ): CostBreakdown {
    // Request cost
    const requestCost = (invocations / 1_000_000) * platform.pricingPer1MRequests

    // Compute cost (billed in 1ms increments, rounded up to 100ms)
    const gbSeconds = (invocations * (avgDurationMs / 1000) * (memoryMB / 1024))
    const computeCost = gbSeconds * platform.pricingPerGBSecond

    return {
      requestCost,
      computeCost,
      totalCost: requestCost + computeCost,
      costPerInvocation: (requestCost + computeCost) / invocations
    }
  }

  findCostOptimalConfiguration(
    requirements: PerformanceRequirements,
    platforms: ServerlessPlatform[]
  ): { platform: ServerlessPlatform; config: DeploymentConfig; estimatedCost: number } {
    let bestOption = null
    let lowestCost = Infinity

    for (const platform of platforms) {
      for (let memory = 128; memory <= platform.maxMemory; memory *= 2) {
        const estimatedDuration = this.estimateDuration(requirements, memory)
        const cost = this.calculateMonthlyCost(
          requirements.monthlyInvocations,
          estimatedDuration,
          memory,
          platform
        )

        if (cost.totalCost < lowestCost && estimatedDuration <= requirements.maxLatencyMs) {
          lowestCost = cost.totalCost
          bestOption = { platform, memory, cost: cost.totalCost }
        }
      }
    }

    return bestOption!
  }

  private estimateDuration(requirements: PerformanceRequirements, memory: number): number {
    // Estimate function duration based on memory allocation
    // More memory = faster CPU = shorter execution
    const baseDuration = requirements.estimatedBaseDurationMs
    return baseDuration / (memory / 128) // Rough estimate
  }
}

interface CostBreakdown {
  requestCost: number
  computeCost: number
  totalCost: number
  costPerInvocation: number
}

interface PerformanceRequirements {
  monthlyInvocations: number
  estimatedBaseDurationMs: number
  maxLatencyMs: number
}
```

## Best Practices

1. **Minimize Package Size**: Reduce cold start time by keeping function size small
2. **Lazy Load Dependencies**: Load heavy libraries only when needed
3. **Implement Caching**: Use in-memory caching across invocations
4. **Idempotent Functions**: Design functions to handle duplicate invocations safely
5. **Error Handling**: Implement comprehensive error handling and retry logic
6. **Environment Variables**: Use for configuration instead of hardcoding
7. **Monitoring**: Set up metrics and alarms for production functions
8. **VPC Caution**: Only use VPC when necessary due to cold start overhead
9. **Cost Monitoring**: Track costs and optimize memory/timeout settings
10. **Security**: Use least-privilege IAM roles and encrypt sensitive data
